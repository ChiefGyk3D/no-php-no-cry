{{- /* Markdown image render hook. Only page-bundle images are accepted (a file beside index.md), so the page never requests a third-party image. Only JPEG and PNG are processed; no WebP. Alt text is required and the build fails without it. Output carries srcset, sizes, width, height and lazy loading. */ -}} {{- $alt := .PlainText | strings.TrimSpace -}} {{- if not $alt -}} {{- errorf "image %q on page %q has no alt text" .Destination .Page.File.Path -}} {{- end -}} {{- $img := .Page.Resources.Get .Destination -}} {{- if not $img -}} {{- errorf "image %q on page %q is not a page-bundle resource (remote and site-wide images are not allowed)" .Destination .Page.File.Path -}} {{- end -}} {{- if not (in (slice "image/jpeg" "image/png") $img.MediaType.Type) -}} {{- errorf "image %q on page %q is %s; only image/jpeg and image/png are supported" .Destination .Page.File.Path $img.MediaType.Type -}} {{- end -}} {{- $widths := slice 480 800 1200 -}} {{- $fallback := $img -}} {{- if gt $img.Width 800 }}{{ $fallback = $img.Resize "800x" }}{{ end -}} {{- $set := slice -}} {{- range $widths -}} {{- if le . $img.Width -}} {{- $r := $img.Resize (printf "%dx" .) -}} {{- $set = $set | append (printf "%s %dw" $r.RelPermalink $r.Width) -}} {{- end -}} {{- end -}}

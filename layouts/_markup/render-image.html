{{- /* Markdown image render hook. Only page-bundle images are accepted (a file beside index.md), so the page never requests a third-party image. Only JPEG, PNG and WebP are processed, plus a page-bundle SVG (issue 352), served as is with the width and height of its viewBox (scripts/check-content.py keeps SVGs inert). Alt text is required and the build fails without it. Output carries srcset, sizes, width, height and lazy loading. */ -}} {{- $alt := .PlainText | strings.TrimSpace -}} {{- if not $alt -}} {{- errorf "image %q on page %q has no alt text" .Destination .Page.File.Path -}} {{- end -}} {{- /* The migration converter writes "TODO alt text" where the source had no alt. Drafts are never rendered, so this only stops a published page. */ -}} {{- if strings.HasPrefix $alt "TODO alt text" -}} {{- errorf "image %q on page %q still has placeholder alt text; write a real description" .Destination .Page.File.Path -}} {{- end -}} {{- $img := .Page.Resources.Get .Destination -}} {{- if not $img -}} {{- errorf "image %q on page %q is not a page-bundle resource (remote and site-wide images are not allowed)" .Destination .Page.File.Path -}} {{- end -}} {{- if eq $img.MediaType.Type "image/svg+xml" -}} {{- /* Only the ROOT start tag is read: its attributes are walked token by token, so a viewBox written inside another attribute's value is never matched. The value must be "0 0 width height", plain decimals, each at most 10000 (scripts/check-content.py enforces the same rule). */ -}} {{- $root := findRESubmatch `^\s*(?:<\?xml[^>]*\?>\s*)? {{- else -}} {{- if not (in (slice "image/jpeg" "image/png" "image/webp") $img.MediaType.Type) -}} {{- errorf "image %q on page %q is %s; only JPEG, PNG and WebP are supported" .Destination .Page.File.Path $img.MediaType.Type -}} {{- end -}} {{- $widths := slice 480 800 1200 -}} {{- $fallback := $img -}} {{- if gt $img.Width 800 }}{{ $fallback = $img.Resize "800x" }}{{ end -}} {{- $set := slice -}} {{- range $widths -}} {{- if le . $img.Width -}} {{- $r := $img.Resize (printf "%dx" .) -}} {{- $set = $set | append (printf "%s %dw" $r.RelPermalink $r.Width) -}} {{- end -}} {{- end -}} {{- end -}}
